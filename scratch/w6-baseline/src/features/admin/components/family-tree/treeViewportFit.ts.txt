interface NodeBounds { x: number; y: number; width: number; height: number }

/** Fit visible content rather than layout padding; height limits keep all generations reachable. */
export function fitTreeContent(nodes: readonly NodeBounds[], viewport: { width: number; height: number }, toolbarHeight = 64) {
  if (!nodes.length || viewport.width <= 0 || viewport.height <= 0) return null;
  const left = Math.min(...nodes.map(node => node.x));
  const right = Math.max(...nodes.map(node => node.x + node.width));
  const top = Math.min(...nodes.map(node => node.y));
  const bottom = Math.max(...nodes.map(node => node.y + node.height));
  const usefulHeight = Math.max(120, viewport.height - toolbarHeight - 40);
  const scale = Math.max(.05, Math.min(viewport.width * .72 / Math.max(1, right - left), usefulHeight / Math.max(1, bottom - top), 1.35));
  return { scale, x: viewport.width / 2 - (left + right) / 2 * scale, y: toolbarHeight + 16 + usefulHeight / 2 - (top + bottom) / 2 * scale };
}
