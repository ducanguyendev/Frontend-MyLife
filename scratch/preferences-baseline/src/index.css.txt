/* Keyboard focus uses a subtle inner cue, never an outer accent-colored ring. */
:where(button, [tabindex], input, select, textarea):not([data-ui-control]):focus-visible {
  outline: none;
  box-shadow: inset 0 0 0 1px var(--focus-border);
}

/* Controls that already have a border use that single border as the focus cue. */
:where(button, [tabindex], input, select, textarea):not([data-ui-control])[class*="border"]:focus-visible {
  border-color: var(--focus-border);
  box-shadow: none;
}

/* Links use a text cue instead of a rounded focus frame. */
:where(a):focus-visible {
  outline: none;
  text-decoration: underline;
  text-decoration-color: var(--focus-border);
  text-underline-offset: 3px;
}
