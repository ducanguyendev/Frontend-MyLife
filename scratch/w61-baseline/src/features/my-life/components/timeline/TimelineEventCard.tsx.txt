import { Link } from 'react-router-dom';
import { CalendarClock, MapPin, Pencil, Trash2, UsersRound } from 'lucide-react';
import { Badge } from '@/shared/components/ui';
import { MemoryActions } from '../memories/MemoryActions';
import type { FamilyTimelineEvent } from '@/features/admin/services/familyV2Service';
import { getTimelineEventTypeTranslationKey } from './eventTypeTranslation';
import { getTimelineEventDisplayTitle } from './timelinePresentation';
import { isDerivedOnlyEventType } from './types';
import type { TFunction } from 'i18next';

export function TimelineEventCard({ event, date, onEdit, onDelete, translate, relatedMemberName }: { event: FamilyTimelineEvent; date: string; onEdit: (event: FamilyTimelineEvent) => void; onDelete: (event: FamilyTimelineEvent) => void; translate: TFunction; relatedMemberName?: string }) {
 const derived = event.isDerived || isDerivedOnlyEventType(event.eventType);
 return <li className="timeline-event" data-derived={derived}>
  <span aria-hidden="true" className="timeline-event__marker"/>
  <p className="timeline-event__date">{date}{event.isApproximate && <CalendarClock size={14} aria-label={translate('admin.family_v2.approximate_date')}/>}</p>
  <article className="timeline-event__body">
   <div className="timeline-event__meta"><Badge variant="outline">{translate(getTimelineEventTypeTranslationKey(event.eventType))}</Badge>
    {derived && <span className="timeline-event__automatic">{translate('memories.automatic')}</span>}
    {!derived && <MemoryActions label={translate('memories.eventActions')} actions={[
     { label:translate('admin.family_v2.edit_event'), icon:<Pencil size={14}/>, onClick:() => onEdit(event) },
     { label:translate('admin.family_v2.delete_event'), icon:<Trash2 size={14}/>, danger:true, onClick:() => onDelete(event) },
    ]}/>}
   </div>
   <h3>{getTimelineEventDisplayTitle(event, translate)}</h3>
   {(event.memberName || relatedMemberName) && <div className="timeline-event__people"><UsersRound size={14}/>
    {event.memberName && (event.memberId ? <Link to={`/mylife/family/member/${event.memberId}`}>{event.memberName}</Link> : <span>{event.memberName}</span>)}
    {relatedMemberName && event.relatedMemberId && <><span aria-hidden="true">·</span><Link to={`/mylife/family/member/${event.relatedMemberId}`}>{relatedMemberName}</Link></>}
   </div>}
   {event.description && <p className="timeline-event__description">{event.description}</p>}
   {event.location && <p className="timeline-event__location"><MapPin size={14} aria-hidden="true"/>{event.location}{event.latitude !== null && event.longitude !== null && <a href={`https://www.openstreetmap.org/?mlat=${event.latitude}&mlon=${event.longitude}#map=14/${event.latitude}/${event.longitude}`} target="_blank" rel="noreferrer">{translate('admin.family_v2.view_on_map')}</a>}</p>}
  </article>
 </li>;
}
